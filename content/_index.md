        <script>
          let currentDiagTab = 'autonomic';

          function switchDiagnosisTab(tab) {
            currentDiagTab = tab;
            const tabs = ['autonomic', 'panic', 'insomnia'];
            tabs.forEach(t => {
              const panel = document.getElementById('diag-panel-' + t);
              const btn = document.getElementById('diag-btn-' + t);
              if (t === tab) {
                if (panel) panel.style.display = 'block';
                if (btn) {
                  btn.className = "diag-tab-btn flex-1 sm:flex-initial px-3 sm:px-4 py-2 sm:py-2.5 rounded-xl font-extrabold text-xs sm:text-sm transition bg-[#2F5D50] text-white shadow-sm cursor-pointer whitespace-nowrap text-center";
                }
              } else {
                if (panel) panel.style.display = 'none';
                if (btn) {
                  btn.className = "diag-tab-btn flex-1 sm:flex-initial px-3 sm:px-4 py-2 sm:py-2.5 rounded-xl font-extrabold text-xs sm:text-sm transition bg-[#F4F7F5] text-[#556B62] hover:bg-[#E8EFEA] border border-[#DDE6E1] cursor-pointer whitespace-nowrap text-center";
                }
              }
            });
            calculateDiagnosis();
          }

          function calculateDiagnosis() {
            const activePanel = document.getElementById('diag-panel-' + currentDiagTab);
            if (!activePanel) return;
            const checkedBoxes = activePanel.querySelectorAll('.diag-checkbox:checked');
            const count = checkedBoxes.length;
            
            const countEl = document.getElementById('diag-count');
            const statusEl = document.getElementById('diag-status');
            const descEl = document.getElementById('diag-desc');
            const cardEl = document.getElementById('diag-result-card');

            if (countEl) countEl.textContent = count;

            if (count === 0) {
              if (statusEl) statusEl.innerHTML = '증상을 1개 이상 체크해 보세요.';
              if (descEl) descEl.textContent = '체크된 증상의 개수에 따라 자율신경계 과흥분 및 뇌 신경계 불균형 여부를 실시간 평가합니다.';
              if (cardEl) {
                cardEl.className = 'p-5 rounded-2xl bg-white border-2 border-[#DDE6E1] shadow-sm transition-all duration-300 space-y-3';
              }
            } else if (count <= 2) {
              if (statusEl) statusEl.innerHTML = '<span class="text-[#2F5D50]">🌱 [경미/주의]</span> 초기 자율신경 스트레스 신호';
              if (descEl) descEl.textContent = '아직 심각하지는 않으나, 피로와 스트레스가 누적되어 신경계 불균형이 시작되는 단계입니다. 충분한 휴식과 수면 관리가 필요합니다.';
              if (cardEl) {
                cardEl.className = 'p-5 rounded-2xl bg-[#F8FAF9] border-2 border-[#2F5D50] shadow-sm transition-all duration-300 space-y-3';
              }
            } else if (count <= 3) {
              if (statusEl) statusEl.innerHTML = '<span class="text-[#B88A3B]">⚠️ [상담 권장]</span> 자율신경계 불균형 진행 단계';
              if (descEl) descEl.textContent = '교감신경 항진 및 신경전달물질의 불균형으로 신체화 증상이 복합 발현되고 있습니다. 만성화되기 전 조기 정밀 검사와 1:1 맞춤 치료를 추천합니다.';
              if (cardEl) {
                cardEl.className = 'p-5 rounded-2xl bg-[#FFFDF5] border-2 border-[#F4D06F] shadow-md transition-all duration-300 space-y-3';
              }
            } else {
              if (statusEl) statusEl.innerHTML = '<span class="text-[#E53E3E]">🚨 [집중 치료 필요]</span> 뇌 신경계 과흥분 및 만성 불균형';
              if (descEl) descEl.textContent = '일상생활에 지장을 줄 만큼 신경계 자생력이 크게 저하된 상태입니다. 뇌파·체열 검사를 통한 근본 원인 분석과 즉각적인 1:1 맞춤 한방 치료가 시급합니다.';
